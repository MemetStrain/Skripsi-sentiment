ALGORITHM reconcile_aggregates
SOURCE: scheduler/reconcile.py :: reconcile_aggregates (dipanggil oleh reconcile_all)

INPUT:
    db       // Firestore client
    paths    // dict berisi 'news_raw' dan 'news_sent'

OUTPUT:
    int      // jumlah dokumen yang ditulis ke D3

BEGIN
    // 3.1 load seluruh artikel dari CSV lokal (termasuk sentimen)
    articles = load_news_from_csv(paths['news_raw'], paths['news_sent'])
                                                     // news_extractor.py:175
    IF articles IS EMPTY THEN
        log("No articles found in CSV; skip aggregation")
        RETURN 0
    END IF

    // 3.2 hitung bucket agregat harian
    aggregates = compute_sentiment_aggregates(articles) // sentiment_runner.py:139
    IF aggregates IS EMPTY THEN
        log("No aggregates computed; skip reconcile")
        RETURN 0
    END IF

    // 3.3a ambil existing keys dari D3
    existing_docs = db.collection('sentiment_aggregates').stream()
    existing_keys = { doc.id FOR doc IN existing_docs }
                                                     // format: "Daily_YYYY-MM-DD"

    // 3.3b bandingkan → filter stale (missing atau berubah)
    stale_aggs = [
        agg FOR agg IN aggregates
        IF (agg.frequency + "_" + agg.date) NOT IN existing_keys
    ]

    IF stale_aggs IS EMPTY THEN
        log("D3 already up-to-date; 0 writes")
        RETURN 0
    END IF

    // 3.3c batch upsert ke D3
    n_written = write_sentiment_aggregates(db, stale_aggs)
                                                     // firestore_writer.py:197
    log("Wrote " + n_written + " aggregates to D3 sentiment_aggregates")
    RETURN n_written
END

FUNCTION compute_sentiment_aggregates (helper 3.2)

FUNCTION compute_sentiment_aggregates(articles) RETURNS list[dict]
SOURCE: scheduler/sentiment_runner.py :: compute_sentiment_aggregates

INPUT:
    articles   // list[dict] — setiap artikel memiliki: date, sentiment_label,
               //             positive_prob, negative_prob, neutral_prob, sentiment_score

BEGIN
    buckets = {}   // key = (frequency, date_str)

    FOR article IN articles DO
        date_str = article.date[0:10]                 // ambil YYYY-MM-DD
        key      = ("Daily", date_str)

        IF key NOT IN buckets THEN
            buckets[key] = {
                frequency       : "Daily",
                date            : date_str,
                article_count   : 0,
                positive_prob   : 0.0,
                negative_prob   : 0.0,
                neutral_prob    : 0.0,
                sentiment_score : 0.0
            }
        END IF

        b = buckets[key]
        b.article_count   += 1
        b.positive_prob   += article.positive_prob   OR 0.0
        b.negative_prob   += article.negative_prob   OR 0.0
        b.neutral_prob    += article.neutral_prob    OR 0.0
        b.sentiment_score += article.sentiment_score OR 0.0
    END FOR

    aggregates = []
    FOR b IN buckets.values() DO
        n = b.article_count
        IF n > 0 THEN
            b.positive_prob   /= n                   // rata-rata probabilitas
            b.negative_prob   /= n
            b.neutral_prob    /= n
            b.sentiment_score /= n
        END IF
        aggregates.append(b)
    END FOR

    RETURN aggregates
END FUNCTION
