ALGORITHM update_hmm_states
SOURCE: scheduler/hmm_updater.py :: update_hmm_states

INPUT:
    db               // Firestore client
    write_existing   // bool = False (default: skip tanggal yang sudah ada di D4)

OUTPUT:
    int              // jumlah dokumen yang ditulis ke D4

BEGIN
    // 4.1 load frozen HMM params dari D5 (hmm_models/Daily)
    params = read_hmm_params(db, 'Daily')            // firestore_writer.py:140
    IF params IS NULL THEN
        log("No HMM model found in D5; skip")
        RETURN 0
    END IF
    feat_cols     = params.feat_cols
    vol_window    = params.vol_window
    norm_window   = params.norm_window
    state_to_label = params.state_to_label           // e.g. {0: "Bearish", 1: "Bullish", 2: "Neutral"}

    // 4.2 load seluruh harga dari D1 (daily_prices), urut ascending
    price_rows = db.collection('daily_prices').order_by('date').stream()
    IF price_rows IS EMPTY THEN
        log("No price data found; skip")
        RETURN 0
    END IF
    price_df = to_dataframe(price_rows)              // kolom: date, close

    // 4.3 bangun feature matrix HMM (5 fitur Z-score)
    feat_df = _build_hmm_features(price_df, vol_window, norm_window)
                                                     // hmm_updater.py:79
    IF feat_df IS EMPTY THEN
        log("Feature matrix empty after NaN drop; skip")
        RETURN 0
    END IF
    X = feat_df[feat_cols].values                    // ndarray shape (T, 5)

    // 4.4 dekode states via online forward filter
    model  = _hmm_from_params(params)                // hmm_updater.py:112
    states = _forward_filter(model, X)               // hmm_updater.py:135 → ndarray[int] shape (T,)

    // 4.5 filter + batch write ke D4
    IF write_existing = FALSE THEN
        existing_ids = { doc.id FOR doc IN db.collection('hmm_states').stream() }
    ELSE
        existing_ids = {}
    END IF

    to_write = []
    FOR i IN 0 .. len(feat_df) - 1 DO
        date_str = feat_df.index[i].strftime("YYYY-MM-DD")
        doc_id   = "Daily_" + date_str
        IF doc_id IN existing_ids THEN CONTINUE
        to_write.append({
            doc_id    : doc_id,
            date      : date_str,
            state     : int(states[i]),
            label     : state_to_label[states[i]],
            frequency : "Daily"
        })
    END FOR

    IF to_write IS EMPTY THEN
        log("All dates already present in D4; 0 writes")
        RETURN 0
    END IF

    n_written = write_hmm_states_batch(db, to_write) // firestore_writer.py:73
    log("Wrote " + n_written + " states to D4 hmm_states")
    RETURN n_written
END

FUNCTION _build_hmm_features (helper 4.3)

FUNCTION _build_hmm_features(price_df, vol_window, norm_window) RETURNS DataFrame
SOURCE: scheduler/hmm_updater.py :: _build_hmm_features

BEGIN
    df = price_df.copy()
    df['Log_Return'] = log(df.close / df.close.shift(1))

    // volatilitas rolling (std log-return)
    df['Volatility']  = df['Log_Return'].rolling(vol_window).std()

    // RSI-14 dinormalisasi ke [-1, +1]
    df['RSI']         = ta.RSI(df.close, timeperiod=14)
    df['RSI_norm']    = (df['RSI'] - 50) / 50

    // MACD = EMA12 - EMA26
    df['MACD'], _, _  = ta.MACD(df.close, fastperiod=12, slowperiod=26, signalperiod=9)

    // Bollinger Band Width = (upper - lower) / middle
    upper, mid, lower = ta.BBANDS(df.close, timeperiod=20)
    df['BB_Width']    = (upper - lower) / mid

    // Z-score tiap fitur dalam rolling window norm_window (default 252d)
    FOR col IN ['Log_Return', 'Volatility', 'MACD', 'BB_Width'] DO
        mu             = df[col].rolling(norm_window).mean()
        sigma          = df[col].rolling(norm_window).std()
        df[col + '_Z'] = (df[col] - mu) / sigma.replace(0, NaN)
    END FOR

    feat_df = df[['Log_Return_Z', 'Volatility_Z', 'RSI_norm', 'MACD_Z', 'BB_Width_Z']].dropna()
    RETURN feat_df
END FUNCTION

FUNCTION _forward_filter (helper 4.4)

Online forward filter: setiap timestep hanya melihat observasi sampai saat itu (O1..t), sehingga state historis tidak berubah saat data baru tiba (berbeda dengan Viterbi yang merevisi seluruh sequence). 

FUNCTION _forward_filter(model, X) RETURNS ndarray[int]
SOURCE: scheduler/hmm_updater.py :: _forward_filter

INPUT:
    model   // GaussianHMM dengan startprob_, transmat_, means_, covars_
    X       // ndarray shape (T, n_features) — feature matrix dari _build_hmm_features

OUTPUT:
    states  // ndarray[int] shape (T,) — state index per timestep

BEGIN
    T       = len(X)
    n_comp  = model.n_components                     // jumlah state HMM
    states  = zeros(T, dtype=int)

    // hitung log-likelihood observasi untuk setiap state: shape (T, n_comp)
    log_B = model._compute_log_likelihood(X)         // multivariate Gaussian per komponen
    B     = exp(log_B)                               // shape (T, n_comp)

    // t = 0: alpha = pi * B(x_0), normalisasi
    alpha    = model.startprob_ * B[0]
    alpha   /= sum(alpha)                            // normalisasi → probabilitas
    states[0] = argmax(alpha)

    // t = 1 .. T-1: forward pass
    FOR t IN 1 .. T - 1 DO
        // alpha_t = normalize( A^T · alpha_{t-1} ⊙ B(x_t) )
        // A = transmat_, shape (n_comp, n_comp); A[i,j] = P(q_t=j | q_{t-1}=i)
        alpha      = (model.transmat_.T @ alpha) * B[t]
        denom      = sum(alpha)
        IF denom > 0 THEN
            alpha /= denom
        ELSE
            alpha = ones(n_comp) / n_comp             // uniform fallback jika underflow
        END IF
        states[t] = argmax(alpha)
    END FOR

    RETURN states
END FUNCTION
