ALGORITHM _step_news
SOURCE: scheduler/main.py :: _step_news

INPUT:
    paths   // dict berisi 'news_raw', 'news_preproc', 'news_sent'

OUTPUT:
    (scored_articles, did_change)   // list[dict], bool

BEGIN
    // 2.1 tentukan cutoff & latest tersimpan
    cutoff = most_recent_trading_day()
    have   = latest_news_date(paths['news_sent'])

    IF have IS NOT NULL AND have >= cutoff THEN
        log("News CSV is current; skip scrape")
        RETURN ([], FALSE)
    END IF

    // 2.2 scrape multi-keyword × halaman (newest-first; stop at cutoff)
    raw = scrape_new_articles(have)
    IF raw IS EMPTY THEN
        log("No new articles scraped")
        RETURN ([], FALSE)
    END IF

    // 2.5a append CSV raw lebih dulu (arsip, sebelum drop empty)
    raw_rows = [article_to_raw_row(a) FOR a IN raw]
    append_news_rows(paths['news_raw'], raw_rows, RAW_FIELDS)

    // 2.3 preprocess (NewsPreprocessor: clean title + content, drop empty)
    cleaned = preprocess_articles(raw)                  // news_extractor.py:81
    IF cleaned IS EMPTY THEN
        log("All scraped articles dropped during preprocessing")
        RETURN ([], FALSE)
    END IF

    cleaned_rows = [article_to_raw_row(a) FOR a IN cleaned]
    append_news_rows(paths['news_preproc'], cleaned_rows, RAW_FIELDS)  // 2.5b

    // 2.4 FinBERT-Tone title-only, batch 16 (sentiment_runner.py:63)
    scored = run_sentiment_on_articles(cleaned)

    sentiment_rows = [article_to_sentiment_row(a) FOR a IN scored]
    append_news_rows(paths['news_sent'], sentiment_rows, SENTIMENT_FIELDS)  // 2.5c

    // 2.6 reconcile CSV → Firestore D2 (batch upsert; idempoten by md5(url))
    n_written = reconcile_news(db, scored)              // reconcile.py:85
    log("News reconcile: " + n_written + " articles upserted to D2 news_articles")

    RETURN (scored, TRUE)
END

FUNCTION scrape_new_articles (helper 2.2)

FUNCTION scrape_new_articles(cutoff_date) RETURNS list[dict]
SOURCE: scheduler/news_extractor.py :: scrape_new_articles

BEGIN
    IF cutoff_date IS NULL THEN RETURN []

    session  = _make_session()                          // retry+backoff, pool 16 conn
    articles = []
    TRY
        FOR start_url IN MPOB_SEARCH_URLS DO   // keyword=cpo, keyword=crude%20palm%20oil
            articles += _scrape_keyword(session, start_url, cutoff_date)
        END FOR
    FINALLY
        session.close()
    END TRY

    // dedup by URL — 2 keyword search sering overlap
    seen   = set()
    unique = []
    FOR a IN articles DO
        IF a.url IS NOT EMPTY AND a.url NOT IN seen THEN
            seen.add(a.url); unique.append(a)
        END IF
    END FOR
    RETURN unique
END FUNCTION


FUNCTION _scrape_keyword(session, start_url, cutoff_date) RETURNS list[dict]
BEGIN
    articles = []
    current_url = start_url
    WHILE current_url IS NOT NULL DO
        resp = session.get(current_url, timeout=(10,30), verify=FALSE)
        IF resp fail THEN BREAK
        cards = BeautifulSoup(resp.text).select("a > div.rounded.shadow-md")
        IF cards IS EMPTY THEN BREAK

        (batch, reached_cutoff) = _parse_cards(cards, cutoff_date)
        IF batch IS NOT EMPTY THEN
            _fetch_contents(batch)                       // ThreadPool 8 workers, fetch body per URL
            articles += batch
        END IF
        IF reached_cutoff THEN BREAK

        next_btn = soup.find("a", rel="next")
        IF next_btn HAS href THEN
            current_url = next_btn.href; sleep(random(0.3, 0.8))
        ELSE
            current_url = NULL
        END IF
    END WHILE
    RETURN articles
END FUNCTION
