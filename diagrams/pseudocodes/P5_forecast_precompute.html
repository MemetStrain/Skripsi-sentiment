ALGORITHM _step_precompute (wrapper try/except)

ALGORITHM _step_precompute
SOURCE: scheduler/main.py :: _step_precompute

INPUT:  db
OUTPUT: bool   // True jika ≥1 doc forecast ditulis

BEGIN
    TRY
        IMPORT precompute_forecasts
        n = precompute_forecasts.precompute_and_write(db)
    CATCH Exception AS e
        log error("Forecast precompute failed: " + e)
        RETURN FALSE          // langkah 1–4 yang sudah commit tidak di-rollback
    END TRY
    RETURN n > 0
END

ALGORITHM precompute_and_write

ALGORITHM precompute_and_write
SOURCE: scheduler/precompute_forecasts.py :: precompute_and_write

INPUT:  db
OUTPUT: int — jumlah dokumen forecast per-titik ditulis ke D8

CONSTANTS:
    PRECOMPUTE_WINDOW_DAYS = 365
    PRECOMPUTE_MAX_HORIZON = 7
    FREQUENCY              = "Daily"

BEGIN
    inference = _import_inference()                   // sys.path += ../prediction

    payload = inference.compute_forecast_trails(
                  db,
                  max_horizon = PRECOMPUTE_MAX_HORIZON,
                  window_days = PRECOMPUTE_WINDOW_DAYS
              )

    trails = payload.trails
    IF trails IS EMPTY THEN
        log warn("compute_forecast_trails returned no trails"); RETURN 0
    END IF

    points = _flatten_trails(trails)                  // nested trail → flat list of dicts
    IF points IS EMPTY THEN
        log warn("Trails contained no points"); RETURN 0
    END IF

    n = write_forecasts(db, points)                   // → D8 (doc id = Daily_h<h>_<anchor>)

    write_forecast_meta(db, {
        frequency          : FREQUENCY,
        generated_at       : payload.generated_at,
        max_horizon        : PRECOMPUTE_MAX_HORIZON,
        window_days        : PRECOMPUTE_WINDOW_DAYS,
        winners_by_horizon : payload.winners,
        configs_by_horizon : payload.configs,
        metrics            : payload.metrics,
        tag_to_config      : payload.tag_to_config,
        horizons           : payload.horizons,
        forward_fan        : payload.forward_fan
    })                                                 // → D9 forecast_meta/Daily

    _delete_legacy_doc(db)                             // hapus forecasts/latest jika ada
    RETURN n
END


FUNCTION _flatten_trails(trails) RETURNS list[dict]
BEGIN
    points = []
    FOR trail IN trails DO
        horizon = int(trail.horizon)
        tag     = str(trail.tag OR "")
        config  = str(trail.config OR "")
        FOR p IN trail.points DO
            points.append({
                frequency       : "Daily",
                horizon         : horizon,
                tag             : tag,
                config          : config,
                anchor_date     : str(p.anchor_date),
                anchor_price    : float(p.anchor_price OR 0.0),
                predicted_date  : str(p.predicted_date),
                predicted_price : float(p.predicted_price),
                log_return      : float(p.log_return OR 0.0)
            })
        END FOR
    END FOR
    RETURN points
END

ALGORITHM compute_forecast_trails (5.2 core)

ALGORITHM compute_forecast_trails
SOURCE: prediction/inference.py :: compute_forecast_trails

INPUT:
    db, max_horizon=7, window_days=365

OUTPUT:
    payload dict { horizons, winners, configs, trails[], forward_fan[],
                   metrics, tag_to_config, generated_at }

BEGIN
    // load winners.json (← D7) — winning ablation tag & config per horizon
    winners_payload = load_winners()
    winners = { int(h): tag FOR (h, tag) IN winners_payload.winners_by_horizon }
    configs = { int(h): cfg FOR (h, cfg) IN winners_payload.configs_by_horizon }

    // 5.1 bangun inference frame (price + sent + HMM + technicals + lags + cyclical + interaksi)
    df = build_inference_frame(db)
    IF df IS EMPTY THEN
        RETURN { horizons:[], winners:{}, configs:{}, trails:[], metrics:{} }
    END IF

    dates_sorted = df.Date.tolist()
    last_idx     = |dates_sorted| - 1
    next_td      = _next_trading_day_after(dates_sorted[last_idx])

    cutoff     = df.Date.max() - window_days
    window_idx = df.index[df.Date >= cutoff].tolist()

    trails = []

    // 5.2 rolling trails: untuk tiap horizon h, prediksi target di anchor+h
    FOR h IN 1..max_horizon DO
        tag = winners.get(h)
        IF tag IS NULL THEN
            log warn("No winner for h=" + h); CONTINUE
        END IF
        TRY
            (model, scaler, feature_cols) = load_model(tag, h, variant="csa")  // joblib + meta.json
        CATCH Exception
            log warn("Failed to load model h=" + h); CONTINUE
        END TRY

        points         = []
        logged_missing = set()
        FOR i IN window_idx DO
            target_idx = i + h
            IF target_idx > last_idx + 1 THEN CONTINUE           // konvergensi semua trail di T+1
            pred_date = dates_sorted[target_idx] IF target_idx <= last_idx ELSE next_td

            row = df.iloc[[i]]
            TRY
                X = select_feature_matrix(row, feature_cols)
            CATCH ValueError AS e
                IF str(e)[:120] NOT IN logged_missing THEN
                    log warn("h=" + h + " feature mismatch: " + e)
                    logged_missing.add(str(e)[:120])
                END IF
                CONTINUE
            END TRY
            IF any(isnan(X)) THEN CONTINUE
            IF scaler IS NOT NULL THEN X = scaler.transform(X)

            log_ret = float(model.predict(X)[0])
            anchor_close = float(row.Close[0])
            pred_price   = float(anchor_close * exp(clip(log_ret, -10, 10)))

            points.append({
                anchor_date    : dates_sorted[i].strftime("YYYY-MM-DD"),
                anchor_price   : round(anchor_close, 2),
                predicted_date : pred_date.strftime("YYYY-MM-DD"),
                predicted_price: round(pred_price, 2),
                log_return     : round(log_ret, 6)
            })
        END FOR
        trails.append({ horizon: h, tag: tag, config: configs.get(h, ""), points: points })
    END FOR

    // 5.3 forward fan: dari anchor terakhir T, prediksi T+1..T+max_horizon
    forward_fan  = []
    anchor_row   = df.iloc[[last_idx]]
    anchor_close = float(anchor_row.Close[0])
    anchor_date  = dates_sorted[last_idx]
    future_days  = []; fd = anchor_date
    FOR _ IN 1..max_horizon DO fd = _next_trading_day_after(fd); future_days.append(fd) END FOR

    FOR h IN 1..max_horizon DO
        tag = winners.get(h); IF tag IS NULL THEN CONTINUE
        TRY
            (model, scaler, feature_cols) = load_model(tag, h, variant="csa")
            X = select_feature_matrix(anchor_row, feature_cols)
        CATCH Exception AS e
            log warn("Forward fan h=" + h + " skipped: " + e); CONTINUE
        END TRY
        IF any(isnan(X)) THEN CONTINUE
        IF scaler IS NOT NULL THEN X = scaler.transform(X)

        log_ret    = float(model.predict(X)[0])
        pred_price = float(anchor_close * exp(clip(log_ret, -10, 10)))
        forward_fan.append({
            horizon         : h,
            tag             : tag,
            config          : configs.get(h, ""),
            anchor_date     : anchor_date.strftime("YYYY-MM-DD"),
            anchor_price    : round(anchor_close, 2),
            predicted_date  : future_days[h-1].strftime("YYYY-MM-DD"),
            predicted_price : round(pred_price, 2),
            log_return      : round(log_ret, 6)
        })
    END FOR

    RETURN {
        horizons      : range(1, max_horizon+1),
        winners       : { str(h): t FOR (h,t) IN winners },
        configs       : { str(h): c FOR (h,c) IN configs },
        trails        : trails,
        forward_fan   : forward_fan,
        metrics       : winners_payload.metrics,
        tag_to_config : winners_payload.tag_to_config,
        generated_at  : datetime.utcnow().isoformat() + "Z"
    }
END

FUNCTION build_inference_frame (5.1 helper)

FUNCTION build_inference_frame(db) RETURNS DataFrame
SOURCE: prediction/inference.py :: build_inference_frame

BEGIN
    price_df = _load_prices(db)                         // ← D1 daily_prices
    IF price_df IS EMPTY THEN RAISE "daily_prices is empty"

    sent_df      = _load_sentiment(db)                  // ← D3 sentiment_aggregates
    hmm_state_df = _load_hmm_state(db)                  // ← D4 hmm_states

    df = add_price_features(price_df)                   // SMA, EMA, BB, RSI, MACD, ATR, OBV, …
    df = add_hmm_derived_features(df)                   // HMM_RSI, HMM_MACD, HMM_Volatility, …

    df.Change_Pct = df.Close.pct_change() * 100

    IF hmm_state_df IS NOT EMPTY THEN
        df = df.merge(hmm_state_df, on="Date", how="left")
        df.HMM_State       = df.HMM_State.fillna(2).astype(int)
        df.HMM_State_Label = df.HMM_State_Label.fillna("Neutral")
        top_states = df.HMM_State_Label.value_counts().head(5).index.tolist()
        FOR state IN top_states DO
            col_name = "HMM_" + state.replace(" ","_").replace("-","_")
            df[col_name] = (df.HMM_State_Label == state).astype(int)
        END FOR
        df = df.drop(columns=["HMM_State_Label"])
    END IF

    IF sent_df IS NOT EMPTY THEN
        df = df.merge(sent_df, on="Date", how="left")
        FOR c IN [Article_Count, Positive_Prob, Negative_Prob, Neutral_Prob, Sentiment_Score] DO
            IF c IN df.columns THEN df[c] = df[c].fillna(0.0)
        END FOR
        IF "Positive_Prob" IN df.columns AND "Confidence" NOT IN df.columns THEN
            df.Confidence = df[[Positive_Prob, Negative_Prob, Neutral_Prob]].max(axis=1).fillna(0.0)
        END IF
    END IF

    df = engineer_all_features(df)                      // cyclical + all lags + interaksi
    RETURN df.sort_values("Date").reset_index()
END
