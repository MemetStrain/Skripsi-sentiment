6.1 — ALGORITHM dashboard(request)

ALGORITHM dashboard
SOURCE: website/web/views.py :: dashboard

INPUT:  request
OUTPUT: HttpResponse (dashboard.html)

BEGIN
    TRY
        db = firestore.client()
    CATCH ValueError
        messages.error(request, "Firebase is not initialized.")
        RETURN render(request, "dashboard.html", _empty_dashboard_ctx())
    END TRY

    cutoff = (today() - 90.days).isoformat()

    // price_list (90 hari OHLCV)
    price_list = []
    FOR doc IN db.collection("daily_prices").where("date",">=",cutoff).order_by("date").stream() DO
        d = doc.to_dict()
        price_list.append({ date, open, high, low, close, volume } from d)
    END FOR

    // state_dict {date → state_label/state}
    state_dict = {}
    FOR doc IN db.collection("hmm_states").where("date",">=",cutoff).order_by("date").stream() DO
        d = doc.to_dict()
        IF d.frequency NOT IN (NULL, "Daily") THEN CONTINUE
        state_dict[d.date] = { state_label: d.state_label OR "Neutral", state: d.state OR 2 }
    END FOR

    // chart_data: row harga + state_label (None bila tidak ada) + numeric state
    chart_data = []
    label_to_int = { "Bearish":0, "Bullish":1, "Neutral":2 }
    FOR row IN price_list DO
        info = state_dict.get(row.date, { state:NULL, state_label:NULL })
        chart_data.append({
            date, actual=row.close, open, high, low, volume,
            state       : label_to_int.get(info.state_label) IF info.state_label ELSE NULL,
            state_label : info.state_label
        })
    END FOR

    // stats: harga terakhir, mean, max, min
    IF price_list NOT EMPTY THEN
        prices = [p.close FOR p IN price_list]
        stats  = { current_price: prices[-1], avg_price: mean(prices), max_price: max(prices),
                   min_price: min(prices), total_days: |prices| }
        latest_date = price_list[-1].date
    ELSE
        stats = { current_price:0, avg_price:0, max_price:0, min_price:0, total_days:0 }
        latest_date = "N/A"
    END IF

    // metrics + winners dari D9 forecast_meta/Daily
    metrics         = { mape:0, smape:0, accuracy:0, best_model:"N/A" }
    winners_payload = {}
    TRY
        meta_doc = db.collection("forecast_meta").document("Daily").get()
        IF meta_doc.exists THEN
            meta_raw        = meta_doc.to_dict()
            winners_payload = json.loads(meta_raw.payload_json OR "{}")
            h1_tag = winners_payload.winners_by_horizon.get("1")
            IF h1_tag THEN
                h1_metrics = winners_payload.metrics[h1_tag]["1"].CSA
                IF h1_metrics THEN
                    metrics = { mape: round(h1_metrics.mape,2), smape: round(h1_metrics.smape,2),
                                accuracy: round(h1_metrics.da,2),
                                best_model: "XGBoost CSA (" + winners_payload.configs_by_horizon.get("1","?") + ")" }
                END IF
            END IF
        END IF
    CATCH Exception
        // page tetap render dengan metrics N/A
    END TRY

    // sentiment_list (90 hari)
    sentiment_list = []
    TRY
        FOR doc IN db.collection("sentiment_aggregates").where("date",">=",cutoff).order_by("date").stream() DO
            d = doc.to_dict()
            IF d.frequency != "Daily" THEN CONTINUE
            sentiment_list.append({ date, positive_prob*100, negative_prob*100, neutral_prob*100, sentiment_score })
        END FOR
    CATCH Exception DO PASS END
    END TRY

    RETURN render("dashboard.html", {
        chart_data, metrics, winners_data=winners_payload,
        sentiment_data=sentiment_list, stats, latest_date,
        page_title="CPO Price Prediction Dashboard"
    })
END

6.2 — ALGORITHM forecasts_api(request)

ALGORITHM forecasts_api
SOURCE: website/web/views.py :: forecasts_api

INPUT:  request    // GET params: max_horizon (1..7), window_days (7..365)
OUTPUT: JsonResponse 200 / 503 / 500

BEGIN
    // clamp params
    TRY   max_horizon = int(request.GET.max_horizon OR 7)
    CATCH max_horizon = 7
    END TRY
    max_horizon = max(1, min(7, max_horizon))

    TRY   window_days = int(request.GET.window_days OR 90)
    CATCH window_days = 90
    END TRY
    window_days = max(7, min(365, window_days))

    TRY
        db = firestore.client()

        meta_doc = db.collection("forecast_meta").document("Daily").get()
        IF NOT meta_doc.exists THEN
            RETURN JsonResponse({ error: "forecast_meta/Daily not found. Run scheduler/main.py to populate forecasts." }, status=503)
        END IF
        meta_raw    = meta_doc.to_dict() OR {}
        meta_payload = json.loads(meta_raw.payload_json OR "{}")

        cutoff = (today() - window_days.days).isoformat()

        // Python-side filter (hindari composite index)
        trails_by_h = {}
        FOR d IN db.collection("forecasts").stream() DO
            x = d.to_dict() OR {}
            IF (x.frequency OR "Daily") != "Daily" THEN CONTINUE
            h = int(x.horizon OR 0)
            IF h < 1 OR h > max_horizon THEN CONTINUE
            pred_date = x.predicted_date OR ""
            IF pred_date IS EMPTY OR pred_date < cutoff THEN CONTINUE
            trails_by_h.setdefault(h, []).append({
                anchor_date, anchor_price=round(float(x.anchor_price OR 0.0),2),
                predicted_date=pred_date, predicted_price=round(float(x.predicted_price OR 0.0),2),
                log_return=round(float(x.log_return OR 0.0),6)
            })
        END FOR

        IF trails_by_h IS EMPTY THEN
            RETURN JsonResponse({ error: "forecasts collection empty. Run scheduler/main.py to populate forecasts." }, status=503)
        END IF

        winners = meta_payload.winners_by_horizon
        configs = meta_payload.configs_by_horizon

        trails = []
        FOR h IN sorted(trails_by_h.keys()) DO
            points = sorted(trails_by_h[h], key=anchor_date)
            trails.append({ horizon: h, tag: winners.get(str(h), ""), config: configs.get(str(h), ""), points })
        END FOR

        RETURN JsonResponse({
            horizons      : meta_payload.horizons OR range(1, max_horizon+1),
            winners       : winners,
            configs       : configs,
            trails        : trails,
            forward_fan   : meta_payload.forward_fan OR [],
            metrics       : meta_payload.metrics    OR {},
            tag_to_config : meta_payload.tag_to_config OR {},
            generated_at  : meta_raw.generated_at OR ""
        }, status=200)
    CATCH Exception AS e
        RETURN JsonResponse({ error: str(e) }, status=500)
    END TRY
END

6.3 — ALGORITHM news(request)

ALGORITHM news
SOURCE: website/web/views.py :: news     [@cache_control(public=True, s_maxage=3600, max_age=60)]

INPUT:  request    // GET params: sentiment (Positive/Negative/Neutral), page
OUTPUT: HttpResponse (news.html)

BEGIN
    db = firestore.client()
    sentiment_filter = request.GET.sentiment
    page_number      = int(request.GET.page OR 1)
    items_per_page   = 9

    // projection-only stream (drop content field besar)
    news_docs = db.collection("news_articles").select([date,title,category,snippet,url,sentiment_label,sentiment_score]).stream()

    news_list        = []
    sentiment_counts = { positive:0, negative:0, neutral:0, total:0 }
    coverage_start   = NULL; coverage_end = NULL

    FOR doc IN news_docs DO
        d = doc.to_dict()
        adate = d.date OR ""
        IF adate THEN
            IF coverage_start IS NULL OR adate < coverage_start THEN coverage_start = adate
            IF coverage_end   IS NULL OR adate > coverage_end   THEN coverage_end   = adate
        END IF

        label = d.sentiment_label OR "Neutral"
        sentiment_counts.total += 1
        IF label = "Positive" THEN sentiment_counts.positive += 1
        ELSE IF label = "Negative" THEN sentiment_counts.negative += 1
        ELSE sentiment_counts.neutral += 1
        END IF

        IF sentiment_filter AND label != sentiment_filter THEN CONTINUE
        news_list.append({ date, title, category, snippet, url, sentiment_label=label,
                           sentiment_score=float(d.sentiment_score OR 0) })
    END FOR

    sort news_list desc by date
    total_news  = |news_list|
    total_pages = max(1, ceil(total_news / items_per_page))
    page_number = clamp(page_number, 1, total_pages)
    start       = (page_number - 1) * items_per_page
    news_page   = news_list[start : start + items_per_page]

    build pagination window (max 5 visible page numbers around current)

    RETURN render("news.html", { news_page, sentiment_counts, current_filter=sentiment_filter,
                                  pagination, coverage_start, coverage_end,
                                  page_title="CPO News & Sentiment" })
END
