ALGORITHM _step_price
SOURCE: scheduler/main.py :: _step_price

INPUT:
    paths   // dict berisi 'cpo_raw' dan 'cpo_preproc' (path CSV)

OUTPUT:
    bool    // True jika CSV berubah (baris baru di-append)

BEGIN
    // 1.1 cutoff & latest tersimpan
    cutoff = most_recent_trading_day()                  // price_fetcher.py
    have   = latest_price_date(paths['cpo_raw'])        // local_csv_writer.py

    IF have IS NOT NULL AND have >= cutoff THEN
        log("Price CSV is current; skip fetch")
        RETURN FALSE
    END IF

    // 1.2 fetch latest OHLCV via investiny
    price = fetch_latest_price()                        // lookback adaptif 90/365/1095 hari
    IF price IS NULL THEN
        log("fetch_latest_price returned None")
        RETURN FALSE
    END IF

    IF have IS NOT NULL AND price.date <= have THEN
        log("Fetched price not newer than CSV; skip")
        RETURN FALSE
    END IF

    // 1.3 append ke CSV lokal (idempoten by date)
    appended = append_price_row_indonesian(paths['cpo_raw'], price)
    IF appended = FALSE THEN
        log("Price already present in CSV (idempotent skip)")
        RETURN FALSE
    END IF

    // 1.4 re-run preprocessing pipeline → engineered CSV
    preprocess_price_csv(paths['cpo_raw'], paths['cpo_preproc'])

    // 1.5 reconcile CSV → Firestore D1 (batch upsert; idempoten)
    csv_prices = load_prices_from_csv(paths['cpo_preproc'])   // list[dict]
    n_written  = reconcile_prices(db, csv_prices)             // reconcile.py:55
    log("Price reconcile: " + n_written + " rows upserted to D1 daily_prices")

    RETURN TRUE
END

FUNCTION fetch_latest_price (helper 1.2)

FUNCTION fetch_latest_price() RETURNS dict | NULL
SOURCE: scheduler/price_fetcher.py :: fetch_latest_price

BEGIN
    end = now()
    data = NULL
    FOR lookback_days IN [90, 365, 1095] DO
        start = end - lookback_days
        TRY
            data = investiny.historical_data(
                investing_id = 992745,                 // CPO FCPO continuous
                from_date    = start.strftime("MM/DD/YYYY"),
                to_date      = end.strftime("MM/DD/YYYY"),
                interval     = "D"
            )
            IF data IS NOT EMPTY THEN BREAK
        CATCH Exception AS e
            log warn; continue                          // retry dengan window lebih lebar
        END TRY
    END FOR

    IF data IS EMPTY THEN RETURN NULL

    norm  = lowercase_keys(data)
    dates  = norm.date; closes = norm.close
    IF dates IS NULL OR closes IS NULL THEN RETURN NULL

    idx = -1                                            // ambil row terakhir (most recent)
    trade_date = parse_to_ISO(dates[idx])
    IF trade_date IS NULL THEN RETURN NULL

    RETURN {
        date   : trade_date,
        open   : float(norm.open[idx]   OR closes[idx]),
        high   : float(norm.high[idx]   OR closes[idx]),
        low    : float(norm.low[idx]    OR closes[idx]),
        close  : float(closes[idx]),
        volume : float(norm.volume[idx] OR 0)
    }
END FUNCTION
